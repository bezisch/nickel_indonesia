Hero snippet input for LLM validation¶

Unlike the villain pipeline, hero snippets are not split into a main (ML-classifier) file plus a supplementary file. There is no hero-specific classifier output to draw a main pool from, so this notebook produces a single input file using the same corpus-reduction procedure as villain_snippets_supplementary_input_LLM.csv: relevance-filtered snippets restricted to hand-coded training articles, hand-coded texts excluded, actor mentions extracted with the NER actor codebook, and near-duplicate entity mentions canonicalized.

# --- Prepare the hero Claude input (single file, no main/supplementary split) ---
import json
import os
import re

import pandas as pd

path = r"C:\Users\charlott\Dropbox (Personal)\Paper_Indonesia_Nickel\ln_training"
LABELED_PATH = r'C:\Users\charlott\Dropbox (Personal)\Paper_Indonesia_Nickel\ln_labeling\labeled_data_n154.xlsx'
SNIPPETS_PATH = os.path.join(path, 'snippets_qual_filter.csv')
CODEBOOK_PATH = os.path.join(path, 'ner_actor_codebook.json')
hero_output_path = os.path.join(path, 'hero_snippets_input_LLM.xlsx')

all_snippets = pd.read_csv(SNIPPETS_PATH)
hand_coded = pd.read_excel(LABELED_PATH, usecols=['Text', 'Source'])
training_ids = set(
    hand_coded['Source'].dropna().astype(str).str.replace(r'\.txt$', '', regex=True)
)

def normalize_text(value):
    return ' '.join(str(value).split()).casefold()

hand_coded_texts = {
    normalize_text(value) for value in hand_coded['Text'].dropna()
}
article_snippets = all_snippets[
    all_snippets['article_id'].astype(str).isin(training_ids)
].copy()
before_text_filter = len(article_snippets)
article_snippets = article_snippets[
    ~article_snippets['snippet'].map(normalize_text).isin(hand_coded_texts)
].copy()

# Match the full-input schema: one row per recognized actor mention.
with open(CODEBOOK_PATH, encoding='utf-8') as file:
    codebook = json.load(file)

def expand_terms(terms):
    expanded = []
    for raw_term in terms:
        for segment in str(raw_term).split(';'):
            for alternative in segment.split(' OR '):
                alternative = alternative.strip()
                if alternative:
                    expanded.append(alternative)
    return expanded

def make_pattern(term):
    term = term.lower()
    if term.endswith('y') and len(term) > 2 and term[-2] not in 'aeiou':
        return r'\b' + re.escape(term[:-1]) + r'(?:y|ies)\b'
    if re.search(r'(?:s|x|ch|sh|z)$', term):
        return r'\b' + re.escape(term) + r'(?:es)?\b'
    return r'\b' + re.escape(term) + r's?\b'

actor_patterns = {}
for record in codebook:
    category = record['actor_label']
    if category.startswith('*'):
        continue
    terms = [
        term for term in expand_terms(record['named_entities'])
        if term.lower() not in {'market', 'environment'}
    ]
    if terms:
        actor_patterns[category] = re.compile(
            '|'.join(make_pattern(term) for term in terms), re.IGNORECASE
        )

# Treat common references to the same real-world actor as one entity.
ACTOR_CANONICAL = {
    'widodo': 'Joko Widodo',
    'jokowi': 'Joko Widodo',
    'VALE': 'vale',
    'joko widodo': 'Joko Widodo',
    'imip': 'Indonesia Morowali Industrial Park',
    'pt imip': 'Indonesia Morowali Industrial Park',
    'indonesia morowali industrial park': 'Indonesia Morowali Industrial Park',
}

def canonicalize_actor(value):
    return ACTOR_CANONICAL.get(str(value).strip().casefold(), str(value).strip())

expanded_rows = []
for row in article_snippets.itertuples(index=False):
    text = str(row.snippet)
    for category, pattern in actor_patterns.items():
        for match in pattern.finditer(text):
            expanded_rows.append({
                'article_id': row.article_id,
                'actor_category': category,
                'matched_entity_text': canonicalize_actor(match.group(0)),
                'snippet_text': text,
            })

before_dedup = len(expanded_rows)
hero_df = pd.DataFrame(expanded_rows).drop_duplicates().reset_index(drop=True)
hero_df.insert(0, 'snippet_id', range(1, len(hero_df) + 1))
hero_df.to_excel(hero_output_path, index=False)

print(f"All relevance-filtered snippets: {len(all_snippets):,}")
print(f"Hand-coded article IDs: {len(training_ids):,}")
print(f"Rows before exact-text filter: {before_text_filter:,}")
print(f"Actor-mention rows before canonicalization/deduplication: {before_dedup:,}")
print(f"Exact duplicate rows removed after actor canonicalization: {before_dedup - len(hero_df):,}")
print(f"Hero input rows: {len(hero_df):,}")
print(f"Unique document IDs: {hero_df['article_id'].nunique():,}")
print(f"Saved to: {hero_output_path}")

chg  = pd.read_csv(DATA_DIR / 'hero_validation_changelog.csv')

# ── 1. Reconstruct the model's original (pre-validation) labels ──────────────
# The snippet file holds FINAL labels only. The changelog also contains
# "confirmed, no change" rows (old == new), so keep only real changes.
changes = chg[chg['old_value'] != chg['new_value']].copy()

snip['hero_validated'] = snip['llm_target_hero']
snip['hero_model']     = snip['llm_target_hero']
snip = snip.set_index('snippet_id')
snip.loc[changes['snippet_id'], 'hero_model'] = changes['old_value'].values
snip = snip.reset_index()

# ── 2. Performance summary (validated label = ground truth) ──────────────────
m, v = snip['hero_model'] == 'YES', snip['hero_validated'] == 'YES'
tp, fp, fn_known = (m & v).sum(), (m & ~v).sum(), (~m & v).sum()
precision = tp / (tp + fp)

# NO-set spot check from the README: 116 sampled, 0 false negatives found
N_SAMPLE, FN_SAMPLE = 116, 0
fn_rate_ub = beta.ppf(0.975, FN_SAMPLE + 1, N_SAMPLE - FN_SAMPLE)  # 95% upper bound

summary = pd.DataFrame({
    'Metric': [
        'Snippet rows coded', 'Model YES (first pass)', 'Validated YES (final)',
        'Confirmed YES (true positives)', 'Downgraded YES→NO (false positives)',
        'Upgraded NO→YES (false negatives found)', 'Precision (YES calls)',
        'False-positive share of YES calls', 'NO rows spot-checked',
        'False negatives in spot-check',
        'False-negative rate in NO set (95% upper bound)',
        'Overall agreement with validated labels (all rows)',
    ],
    'Value': [
        f'{len(snip):,}', int(m.sum()), int(v.sum()), int(tp), int(fp), int(fn_known),
        f'{precision:.1%}', f'{fp/m.sum():.1%}', N_SAMPLE, FN_SAMPLE,
        f'≤ {fn_rate_ub:.1%}', f'{(snip.hero_model == snip.hero_validated).mean():.2%}',
    ],
})
print('=== Model performance: Claude Sonnet 5 vs. manual validation ===')
print(summary.to_string(index=False))

# Which rows were changed
changed_rows = (changes.merge(snip[['snippet_id', 'matched_entity_text', 'actor_category']],
                              on='snippet_id', how='left')
                [['snippet_id', 'matched_entity_text', 'actor_category',

ADD_DIR = Path('addition_Indonesian_workers_all_roles')

# 1) drop rows the flag says are already in the original input
already = (add['already_in_existing_input'].astype(str).str.strip().str.lower()
           .isin(['true', 'yes', 'y', '1']))
add_new = add[~already].copy()

# 2) safety net: also drop exact content duplicates of the original set
#    (snippet_ids may not be comparable across the two files, so match on content)
def content_key(d):
    return (d['article_id'].astype(str) + '|' + d['matched_entity_text'].astype(str) + '|' +
            d['snippet_text'].astype(str).str.strip())

n_before = len(add_new)
add_new = add_new[~content_key(add_new).isin(set(content_key(snip)))]
print(f'Additional rows: {len(add):,} → {len(add_new):,} after removing '
      f'{already.sum():,} flagged + {n_before - len(add_new):,} content duplicates')

# ── Combine ──────────────────────────────────────────────────────────────────
SRC_ORIG, SRC_ADD = 'Original (manually validated)', 'Additional Indonesian workers (not validated)'

# ── Combine ──────────────────────────────────────────────────────────────────
orig = snip.assign(hero_final=snip['hero_validated'])
new  = add_new.assign(hero_final=add_new['llm_target_hero'].astype(str).str.strip().str.upper())

shared = ['snippet_id', 'article_id', 'actor_category', 'matched_entity_text',
          'snippet_text', 'llm_relevant_problem', 'hero_final', 'llm_evidence', 'llm_reason']
combined = pd.concat([orig[shared], new[shared]], ignore_index=True)
print(f'Combined: {len(combined):,} rows; hero = YES: {(combined.hero_final == "YES").sum()}')
