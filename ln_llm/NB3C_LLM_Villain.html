# Rebuild from all-article relevance-filtered snippets, not villain_actor_role_long.csv.
import json
import re

LABELED_PATH = r'C:\Users\charlott\Dropbox (Personal)\Paper_Indonesia_Nickel\ln_labeling\labeled_data_n154.xlsx'
SNIPPETS_PATH = os.path.join(path, 'snippets_qual_filter.csv')
CODEBOOK_PATH = os.path.join(path, 'ner_actor_codebook.json')
supplementary_path = os.path.join(
    path, 'villain_snippets_supplementary_input_LLM.csv'
)

all_snippets = pd.read_csv(SNIPPETS_PATH)
hand_coded = pd.read_excel(LABELED_PATH, usecols=['Text', 'Source'])
training_ids = set(
    hand_coded['Source'].dropna().astype(str).str.replace(r'\.txt$', '', regex=True)
)

def normalize_text(value):
    return ' '.join(str(value).split()).casefold()

hand_coded_texts = {
    normalize_text(value) for value in hand_coded['Text'].dropna()
}
article_snippets = all_snippets[
    all_snippets['article_id'].astype(str).isin(training_ids)
].copy()
before_text_filter = len(article_snippets)
article_snippets = article_snippets[
    ~article_snippets['snippet'].map(normalize_text).isin(hand_coded_texts)
].copy()

# Match the full-input schema: one row per recognized actor mention.
with open(CODEBOOK_PATH, encoding='utf-8') as file:
    codebook = json.load(file)

def expand_terms(terms):
    expanded = []
    for raw_term in terms:
        for segment in str(raw_term).split(';'):
            for alternative in segment.split(' OR '):
                alternative = alternative.strip()
                if alternative:
                    expanded.append(alternative)
    return expanded

def make_pattern(term):
    term = term.lower()
    if term.endswith('y') and len(term) > 2 and term[-2] not in 'aeiou':
        return r'\b' + re.escape(term[:-1]) + r'(?:y|ies)\b'
    if re.search(r'(?:s|x|ch|sh|z)$', term):
        return r'\b' + re.escape(term) + r'(?:es)?\b'
    return r'\b' + re.escape(term) + r's?\b'

actor_patterns = {}
for record in codebook:
    category = record['actor_label']
    if category.startswith('*'):
        continue
    terms = [
        term for term in expand_terms(record['named_entities'])
        if term.lower() not in {'market', 'environment'}
    ]
    if terms:
        actor_patterns[category] = re.compile(
            '|'.join(make_pattern(term) for term in terms), re.IGNORECASE
        )

# Treat common references to the same Indonesian president as one actor.
ACTOR_CANONICAL = {
    'widodo': 'Joko Widodo',
    'jokowi': 'Joko Widodo',
    'joko widodo': 'Joko Widodo',
}

def canonicalize_actor(value):
    return ACTOR_CANONICAL.get(str(value).strip().casefold(), str(value).strip())

expanded_rows = []
for row in article_snippets.itertuples(index=False):
    text = str(row.snippet)
    for category, pattern in actor_patterns.items():
        for match in pattern.finditer(text):
            expanded_rows.append({
                'article_id': row.article_id,
                'actor_category': category,
                'matched_entity_text': canonicalize_actor(match.group(0)),
                'snippet_text': text,
            })

before_dedup = len(expanded_rows)
supplementary_df = pd.DataFrame(expanded_rows).drop_duplicates().reset_index(drop=True)
supplementary_df.insert(0, 'snippet_id', range(1, len(supplementary_df) + 1))
supplementary_df.to_csv(supplementary_path, index=False)

print(f"All relevance-filtered snippets: {len(all_snippets):,}")
print(f"Hand-coded article IDs: {len(training_ids):,}")
print(f"Rows before exact-text filter: {before_text_filter:,}")
print(f"Actor-mention rows before canonicalization/deduplication: {before_dedup:,}")
print(f"Exact duplicate rows removed after actor canonicalization: {before_dedup - len(supplementary_df):,}")
print(f"Supplementary rows: {len(supplementary_df):,}")
print(f"Unique document IDs: {supplementary_df['article_id'].nunique():,}")
print(f"Saved to: {supplementary_path}")

from pathlib import Path

VILLAIN_DIR = Path('villain_full_after_validation')          # <-- folder with the 1_ and 2_ files (adjust)
ADD_DIR     = Path('addition_Indonesian_workers_all_roles')  # relative to the notebook (ln_llm)

# ── 1. Load one validation batch and reconstruct the model's original labels ─
def load_batch(n, out_file, log_file, list_file):
    d   = pd.read_csv(VILLAIN_DIR / out_file)
    log = pd.read_csv(VILLAIN_DIR / log_file)
    lst = pd.read_csv(VILLAIN_DIR / list_file)

    d['villain_validated'] = d['llm_target_villain'].astype(str).str.strip().str.upper()
    d['villain_model']     = d['villain_validated']
    # the changelog also holds "re-checked, confirmed" rows (old == new): harmless to apply
    d = d.set_index('snippet_id')
    d.loc[log['snippet_id'], 'villain_model'] = log['old_value'].values
    d = d.reset_index()

    # cross-check: the *_full_villain_list file must be exactly the validated YES rows
    assert set(lst['snippet_id']) == set(d.loc[d.villain_validated == 'YES', 'snippet_id']), \
        f'Batch {n}: villain list does not match YES rows in the output file'

    d['batch'] = n
    d['uid']   = f'b{n}_' + d['snippet_id'].astype(str)      # snippet_id restarts at 1 in each batch
    return d

# ── 2. De-duplicate across batches (exact same article + entity + text) ──────
def content_key(d):
    return (d['article_id'].astype(str) + '|' + d['matched_entity_text'].astype(str) + '|' +
            d['snippet_text'].astype(str).str.strip())

both = pd.concat([b1, b2], ignore_index=True)

print(f'Rows batch 1: {len(b1):,} | batch 2: {len(b2):,} | exact duplicates removed: {dups.sum():,}')

# ── 3. Performance summary per batch + combined ──────────────────────────────
def perf(d):
    mod, fin = d['villain_model'], d['villain_validated']
    n_yes, n_unc = (mod == 'YES').sum(), (mod == 'UNCLEAR').sum()
    tp, fp = ((mod == 'YES') & (fin == 'YES')).sum(), ((mod == 'YES') & (fin == 'NO')).sum()
    u_yes, u_no = ((mod == 'UNCLEAR') & (fin == 'YES')).sum(), ((mod == 'UNCLEAR') & (fin == 'NO')).sum()
    no_yes = ((mod == 'NO') & (fin == 'YES')).sum()
    return pd.Series({
        'Snippet rows coded': f'{len(d):,}',
        'Model YES (first pass)': n_yes,
        'Model UNCLEAR (first pass)': n_unc,
        'Validated YES (final)': (fin == 'YES').sum(),
        'YES confirmed (true positives)': tp,
        'YES → NO (false positives)': fp,
        'UNCLEAR → YES': u_yes,
        'UNCLEAR → NO': u_no,
        'NO → YES': no_yes,
        'Precision of explicit YES calls': f'{tp / n_yes:.1%}',
        'False-positive share of YES calls': f'{fp / n_yes:.1%}',
        'Share of UNCLEAR resolved to YES': f'{u_yes / n_unc:.1%}' if n_unc else 'n/a',
        'Agreement with validated (all rows)': f'{(mod == fin).mean():.2%}',
    })

summary = pd.DataFrame({'Batch 1 (main)': perf(both[both.batch == 1]),
