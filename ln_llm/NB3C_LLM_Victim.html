Victim snippet input for LLM validation¶

Unlike the villain pipeline, victim snippets are not split into a main (ML-classifier) file plus a supplementary file. There is no victim-specific classifier output to draw a main pool from, so this notebook produces a single input file using the same corpus-reduction procedure as villain_snippets_supplementary_input_LLM.csv: relevance-filtered snippets restricted to hand-coded training articles, hand-coded texts excluded, actor mentions extracted with the NER actor codebook, and near-duplicate entity mentions canonicalized.

# --- Prepare the victim Claude input (single file, no main/supplementary split) ---
import json
import os
import re

import pandas as pd

path = r"C:\Users\charlott\Dropbox (Personal)\Paper_Indonesia_Nickel\ln_training"
LABELED_PATH = r'C:\Users\charlott\Dropbox (Personal)\Paper_Indonesia_Nickel\ln_labeling\labeled_data_n154.xlsx'
SNIPPETS_PATH = os.path.join(path, 'snippets_qual_filter.csv')
CODEBOOK_PATH = os.path.join(path, 'ner_actor_codebook.json')
victim_output_path = os.path.join(path, 'victim_snippets_input_LLM.xlsx')

all_snippets = pd.read_csv(SNIPPETS_PATH)
hand_coded = pd.read_excel(LABELED_PATH, usecols=['Text', 'Source'])
training_ids = set(
    hand_coded['Source'].dropna().astype(str).str.replace(r'\.txt$', '', regex=True)
)

def normalize_text(value):
    return ' '.join(str(value).split()).casefold()

hand_coded_texts = {
    normalize_text(value) for value in hand_coded['Text'].dropna()
}
article_snippets = all_snippets[
    all_snippets['article_id'].astype(str).isin(training_ids)
].copy()
before_text_filter = len(article_snippets)
article_snippets = article_snippets[
    ~article_snippets['snippet'].map(normalize_text).isin(hand_coded_texts)
].copy()

# Match the full-input schema: one row per recognized actor mention.
with open(CODEBOOK_PATH, encoding='utf-8') as file:
    codebook = json.load(file)

def expand_terms(terms):
    expanded = []
    for raw_term in terms:
        for segment in str(raw_term).split(';'):
            for alternative in segment.split(' OR '):
                alternative = alternative.strip()
                if alternative:
                    expanded.append(alternative)
    return expanded

def make_pattern(term):
    term = term.lower()
    if term.endswith('y') and len(term) > 2 and term[-2] not in 'aeiou':
        return r'\b' + re.escape(term[:-1]) + r'(?:y|ies)\b'
    if re.search(r'(?:s|x|ch|sh|z)$', term):
        return r'\b' + re.escape(term) + r'(?:es)?\b'
    return r'\b' + re.escape(term) + r's?\b'

actor_patterns = {}
for record in codebook:
    category = record['actor_label']
    if category.startswith('*'):
        continue
    terms = [
        term for term in expand_terms(record['named_entities'])
        if term.lower() not in {'market', 'environment'}
    ]
    if terms:
        actor_patterns[category] = re.compile(
            '|'.join(make_pattern(term) for term in terms), re.IGNORECASE
        )

# Treat common references to the same real-world actor as one entity.
ACTOR_CANONICAL = {
    'widodo': 'Joko Widodo',
    'jokowi': 'Joko Widodo',
    'VALE': 'vale',
    'joko widodo': 'Joko Widodo',
    'imip': 'Indonesia Morowali Industrial Park',
    'pt imip': 'Indonesia Morowali Industrial Park',
    'indonesia morowali industrial park': 'Indonesia Morowali Industrial Park',
}

def canonicalize_actor(value):
    return ACTOR_CANONICAL.get(str(value).strip().casefold(), str(value).strip())

expanded_rows = []
for row in article_snippets.itertuples(index=False):
    text = str(row.snippet)
    for category, pattern in actor_patterns.items():
        for match in pattern.finditer(text):
            expanded_rows.append({
                'article_id': row.article_id,
                'actor_category': category,
                'matched_entity_text': canonicalize_actor(match.group(0)),
                'snippet_text': text,
            })

before_dedup = len(expanded_rows)
victim_df = pd.DataFrame(expanded_rows).drop_duplicates().reset_index(drop=True)
victim_df.insert(0, 'snippet_id', range(1, len(victim_df) + 1))
victim_df.to_excel(victim_output_path, index=False)

print(f"All relevance-filtered snippets: {len(all_snippets):,}")
print(f"Hand-coded article IDs: {len(training_ids):,}")
print(f"Rows before exact-text filter: {before_text_filter:,}")
print(f"Actor-mention rows before canonicalization/deduplication: {before_dedup:,}")
print(f"Exact duplicate rows removed after actor canonicalization: {before_dedup - len(victim_df):,}")
print(f"Victim input rows: {len(victim_df):,}")
print(f"Unique document IDs: {victim_df['article_id'].nunique():,}")
print(f"Saved to: {victim_output_path}")

Supplementary file: newly recognized "Indonesian worker" mentions (hero + villain + victim)¶

The actor NER codebook's "Indonesian local community & workers" category previously had no entry for the literal phrase "Indonesian worker(s)" (only "Indonesian nationals", "local workers", "local miners", etc.), so snippets that name Indonesian victims only via that exact phrase - e.g. "Four Chinese and nine Indonesian workers died instantly" - produced no Indonesian-side actor mention at all, even though a Chinese-side mention (e.g. "Chinese nationals") was correctly detected in the same snippet. ner_actor_codebook.json was updated to add Indonesian worker, Indonesian laborer, Indonesian labourer, Indonesian personnel to that category.

Two matching passes, kept distinct via match_type:

  • exact_phrase: the four codebook terms above, same literal-phrase matching as every other actor category.
  • proximity: casualty reporting often elides the repeated noun in coordination - "Ten Indonesian and eight Chinese workers were killed" - or states the country name instead of the adjective, in either word order - "workers from China and Indonesia". No literal phrase covers these, so this pass instead looks for "indonesia(n/ns)" within 50 characters of "worker(s)", in either order, not crossing a sentence boundary. Spot-checked against a 25-row random sample of corpus-wide hits: the large majority are genuine Indonesian- worker mentions, with a handful of looser/ambiguous cases (e.g. "workers employed in Indonesia", which could in principle refer to non-Indonesian staff) - lower precision than exact_phrase, flagged as such rather than silently mixed in, so reviewers can weight it accordingly. Two of the harder examples that prompted this ("46 workers were injured, including 14 Chinese nationals"; "Forty-one workers ..., including 11 Chinese nationals") have no Indonesia/Indonesian word anywhere in the sentence - nationality is only implied by subtraction from a stated total, which neither pass can safely recover without inferring unstated facts.

This cell lives here but is not victim-specific: hero, victim, and villain's supplementary input all draw from the same 151-article hand-coded training pool, but villain's main input (villain_snippets_full_input_LLM.csv) is drawn separately from a much larger, 837-article ML-classified pool. To cover candidates for all three roles, this reruns both passes over the full relevance-filtered corpus (snippets_qual_filter.csv, all articles, only exact hand-coded training texts excluded - no training-article restriction), then checks each match against all four existing inputs (heroe_snippets_input_LLM.xlsx, victim_snippets_input_LLM.xlsx, villain_snippets_full_input_LLM.csv, villain_snippets_supplementary_input_LLM.csv) and flags (does not drop) rows already present there. Saved as indonesian_workers_snippets_input_LLM.xlsx.

# --- Supplementary input across all three roles: actor mentions newly recognized by the
# Indonesian-worker codebook fix (see markdown above). Rebuilt from the full relevance-
# filtered corpus, not article_snippets above, since villain's main pool is not restricted
# to the training articles.
NEW_TERMS = ['Indonesian worker', 'Indonesian laborer', 'Indonesian labourer', 'Indonesian personnel']
NEW_CATEGORY = 'Indonesian local community & workers'
new_terms_output_path = os.path.join(path, 'indonesian_workers_snippets_input_LLM.xlsx')

full_corpus = pd.read_csv(SNIPPETS_PATH)
candidate_snippets = full_corpus[
    ~full_corpus['snippet'].map(normalize_text).isin(hand_coded_texts)
].copy()

# Pass 1: exact phrase, same mechanism as every other actor category.
exact_pattern = re.compile(
    '|'.join(make_pattern(term) for term in NEW_TERMS), re.IGNORECASE
)

# Pass 2: proximity. Catches coordination where the noun is elided ("Ten Indonesian and
# eight Chinese workers") or the country name is used instead of the adjective, in either
# order ("workers from China and Indonesia"). [^.]{0,50} bounds the gap to roughly one
# clause and never crosses a sentence-ending period.
proximity_forward = re.compile(r'\bindonesia(?:ns?|n)?\b[^.]{0,50}?\bworkers?\b', re.IGNORECASE)
proximity_backward = re.compile(r'\bworkers?\b[^.]{0,50}?\bindonesia(?:ns?|n)?\b', re.IGNORECASE)

matched_rows = []
for row in candidate_snippets.itertuples(index=False):
    text = str(row.snippet)
    for match in exact_pattern.finditer(text):
        matched_rows.append({
            'article_id': row.article_id, 'actor_category': NEW_CATEGORY,
            'matched_entity_text': canonicalize_actor(match.group(0)),
            'snippet_text': text, 'match_type': 'exact_phrase',
        })
    seen_spans = set()
    for pattern in (proximity_forward, proximity_backward):
        for match in pattern.finditer(text):
            if match.group(0) in seen_spans:
                continue
            seen_spans.add(match.group(0))
            matched_rows.append({
                'article_id': row.article_id, 'actor_category': NEW_CATEGORY,
                'matched_entity_text': canonicalize_actor(match.group(0)),
                'snippet_text': text, 'match_type': 'proximity',
            })

raw_matched_rows = len(matched_rows)
# Where both passes catch the same span (e.g. plain "Indonesian workers" trivially also
# satisfies the proximity regex with a zero-length gap), keep the exact_phrase row: sort it
# first, then dedup on everything except match_type so only one survives.
new_terms_df = (
    pd.DataFrame(matched_rows)
    .sort_values('match_type')  # 'exact_phrase' < 'proximity' alphabetically
    .drop_duplicates(subset=['article_id', 'actor_category', 'matched_entity_text', 'snippet_text'])
    .reset_index(drop=True)
)
rows_before_span_collapse = len(new_terms_df)

# A single snippet can trigger more than one span (e.g. an exact_phrase hit and a separate
# proximity hit, or two distinct proximity hits further apart in a longer snippet) - these
# are the same underlying actor mention for review purposes, so collapse to one row per
# (article_id, actor_category, snippet_text, match_type), keeping the longest/most
# informative matched_entity_text among the collapsed spans.
new_terms_df = (
    new_terms_df
    .assign(_span_len=new_terms_df['matched_entity_text'].str.len())
    .sort_values('_span_len', ascending=False)
    .drop_duplicates(subset=['article_id', 'actor_category', 'snippet_text', 'match_type'])
    .drop(columns='_span_len')
    .reset_index(drop=True)
)

# Flag (do not drop) rows already present in any of the three pipelines' existing input
# files: victim_snippets_input_LLM.xlsx was regenerated earlier today with the fixed
# codebook already in it (a side effect of testing, not deliberate), so it is no longer a
# trustworthy already-reviewed baseline - these rows were never in the version anyone
# actually reviewed. Kept in, with a flag, rather than silently excluded.
existing_frames = {
    'hero': pd.read_excel(os.path.join(path, 'heroe_snippets_input_LLM.xlsx')),
    'victim': victim_df,
    'villain_full': pd.read_csv(os.path.join(path, 'villain_snippets_full_input_LLM.csv')),
    'villain_supplementary': pd.read_csv(os.path.join(path, 'villain_snippets_supplementary_input_LLM.csv')),
}
KEY_COLS = ['article_id', 'actor_category', 'matched_entity_text', 'snippet_text']
existing_keys = set()
for frame in existing_frames.values():
    cols = [c for c in KEY_COLS if c in frame.columns]
    existing_keys |= set(frame[cols].apply(tuple, axis=1))

new_keys = new_terms_df[KEY_COLS].apply(tuple, axis=1)
already_present = new_keys.isin(existing_keys)
new_terms_df['already_in_existing_input'] = already_present.values
new_terms_df.insert(0, 'snippet_id', range(1, len(new_terms_df) + 1))

new_terms_df.to_excel(new_terms_output_path, index=False)

print(f"Full relevance-filtered corpus: {len(full_corpus):,}")
print(f"After excluding exact hand-coded training texts: {len(candidate_snippets):,}")
print(f"Raw matches, both passes, before dedup: {raw_matched_rows:,}")
print(f"After removing exact duplicate (article/entity/snippet) rows: {rows_before_span_collapse:,}")
print(f"After collapsing multiple spans within the same snippet to one row: {len(new_terms_df):,}")
print(new_terms_df['match_type'].value_counts().rename('rows by match_type').to_string())
print(f"Already present in an existing hero/victim/villain input file (flagged, not dropped): {already_present.sum()}")
print(f"Total rows kept (all): {len(new_terms_df):,}")
print(f"Unique articles: {new_terms_df['article_id'].nunique():,}")
print(f"Saved to: {new_terms_output_path}")
